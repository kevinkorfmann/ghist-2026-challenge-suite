def test_diploid_state_partitions_true_two_haplotype_sharing(tmp_path):
    map_path = tmp_path / 'map'
    map_path.write_text('1 . 0 1\n1 . 100 101\n')
    segments = tmp_path / 'ibd.gz'
    with gzip.open(segments, 'wt') as handle:
        handle.write('a 1 b 1 1 1 101 100\n')
        handle.write('a 2 b 2 1 21 81 60\n')
    assert detected_segments(segments, map_path, 'raw_haplotype')[('a', 'b')] == [('1', 100.0), ('1', 60.0)]
    assert detected_segments(segments, map_path, 'diploid_state')[('a', 'b')] == [
        ('1', 20.0), ('1', 60.0), ('1', 60.0), ('1', 20.0)]

def _synthetic_parents(depths):
    parents = {'root1': (), 'root2': ()}
    identifiers = {}
    for branch, depth in enumerate(depths, 1):
        previous = ('root1', 'root2')
        for generation in range(2, depth + 2):
            person = f'validation1_g{generation}-b{branch}-i1'
            sibling = f'validation1_g{generation}-b{branch}-i2'
            spouse = f'mate_{branch}_{generation}'
            parents[person] = previous
            parents[sibling] = previous
            parents[spouse] = ()
            previous = (person, spouse)
        identifiers[person] = f'sample_{len(identifiers)}'
        identifiers[sibling] = f'sample_{len(identifiers)}'
    return parents, identifiers


def test_two_whole_family_designs_cover_all_task_labels_without_pair_exclusions():
    from ghist_relatedness.labels import HUMAN_LABELS

        parents, identifiers = _synthetic_parents(topology['depths'])
        truth = family_truth(parents, identifiers)
        n = len(identifiers)
        assert len(truth) == n * (n - 1) // 2
        labels.update(row['relation'] for row in truth)
    assert labels == set(HUMAN_LABELS.values())

def test_global_tree_recovers_additive_sibling_and_cousin_structure():
    import itertools

    from ghist_relatedness.labels import HUMAN_LABELS
    from scripts.develop_task10_detected_ibd import global_tree_predictions
    features, predictions = [], []
    for i, j in itertools.combinations(range(6), 2):
        label = 'siblings' if i // 2 == j // 2 else 'full_1st_cousin'
        rel = 0.5 if label == 'siblings' else 0.125
        pair = {'indID1': f'a{i}', 'indID2': f'a{j}'}
        features.append({**pair, 'relatedness_ge_5': rel})
        probabilities = {name: 0.05 / 13 for name in HUMAN_LABELS.values()}
        probabilities[label] = 0.95
        predictions.append({**pair, 'relation': label, 'probabilities': probabilities})
    result, diagnostics = global_tree_predictions(features, predictions)
    assert diagnostics['state'] == 'COMPLETED'
    assert [row['relation'] for row in result] == [row['relation'] for row in predictions]
    assert len(result) == 15
    with pytest.raises(ValueError, match='every unique'):
        global_tree_predictions(features[:-1], predictions)

def test_phased_scores_are_invariant_to_homolog_and_sample_relabeling():

    from scripts.develop_task10_detected_ibd import phase_pair_features
    genetic_map = {'1': (np.array([1, 101]), np.array([0., 100.])),
                   '2': (np.array([1, 101]), np.array([0., 100.]))}

    assert phase_pair_features(swapped, genetic_map, 200) == expected
    assert expected['phase_minor_high_5'] == pytest.approx(30 / 120)
    assert expected['phase_minor_low_5'] == 0

def test_multiway_components_are_invariant_and_reject_conflicting_homolog_merges():
    import numpy as np
    module = _multiway_module()
    genetic_map = {'1': (np.array([1, 101]), np.array([0., 100.]))}
    segments = {('a', 'b'): {'1': [(1, 101, 1, 1), (1, 101, 1, 1)]},
                ('a', 'c'): {'1': [(1, 101, 1, 2)]},
                ('b', 'c'): {'1': [(1, 101, 1, 2)]}}
    expected = module.component_integrals([['a'], ['b'], ['c']], segments, genetic_map, 100, 5)
    assert expected == {'union': 1., 'copies': 1., 'multiple': 0., 'conflict': 0., 'complete': 1.}
    flipped = {pair: {chrom: [(lo, hi, 3 - h1 if pair[0] == 'a' else h1,
                3 - h2 if pair[1] == 'a' else h2) for lo, hi, h1, h2 in rows]
            for chrom, rows in chromosomes.items()} for pair, chromosomes in segments.items()}
    assert module.component_integrals([['c'], ['a'], ['b']], flipped, genetic_map, 100, 5) == expected
    segments[('a', 'c')]['1'].append((1, 101, 2, 2))
    conflict = module.component_integrals([['a'], ['b'], ['c']], segments, genetic_map, 100, 5)
    assert conflict['conflict'] == 1.
    assert conflict['union'] == conflict['copies'] == conflict['complete'] == 0.

@pytest.mark.parametrize('engine_name', ['original', 'fast'])
def test_complete_context_features_ignore_sample_names_order_and_homolog_labels(engine_name, monkeypatch):
    import itertools

    import numpy as np
    module = _multiway_module()
    original_module = module
    if engine_name == 'fast':
        from pathlib import Path
        monkeypatch.syspath_prepend(str(Path(__file__).parents[1] / 'experiments/task10_development_2026_11_20'))
        import fast_multiway_features
        module = fast_multiway_features

    class FixedSiblingDetector:
        classes_ = np.array(['siblings', 'full_1st_cousin'])

        def predict_proba(self, matrix):
            return np.array([[.99, .01] if row[0] >= .1 else [.01, .99] for row in matrix])

    model = {'model': FixedSiblingDetector(), 'feature_names': ['ibd2_fraction_5']}
    genetic_map = {'1': (np.array([1, 101]), np.array([0., 100.]))}
    samples = ['a', 'b', 'c', 'd', 'e']
    segments = {('a', 'b'): {'1': [(1, 101, 1, 1), (1, 101, 2, 2)]},
                ('c', 'd'): {'1': [(1, 101, 1, 1), (1, 101, 2, 2)]}}
    for a, b in itertools.combinations(samples, 2):
        if (a, b) not in segments:
            segments[(a, b)] = {'1': [(1, 51, 1, 1)]}
    original, groups, _ = module.family_features(samples, segments, genetic_map, 100, model)
    if engine_name == 'fast':
        reference, reference_groups, _ = original_module.family_features(samples, segments, genetic_map, 100, model)
        assert groups == reference_groups
        for left, right in zip(original, reference, strict=True):
            assert {key: value for key, value in left.items() if key not in ['indID1', 'indID2']} == pytest.approx(
                {key: value for key, value in right.items() if key not in ['indID1', 'indID2']}, abs=1e-12, rel=1e-12)
    assert groups == [('a', 'b'), ('c', 'd'), ('e',)]
    rename = dict(zip(samples, reversed(samples), strict=True))
    transformed = {}
    for (a, b), chromosomes in segments.items():
        x, y = rename[a], rename[b]
        rows = [(lo, hi, 3 - h1 if a in ['a', 'c'] else h1,
                 3 - h2 if b in ['a', 'c'] else h2) for lo, hi, h1, h2 in chromosomes['1']]
        if x > y:
            x, y = y, x
            rows = [(lo, hi, h2, h1) for lo, hi, h1, h2 in rows]
        transformed[(x, y)] = {'1': rows}
    repeated, _, _ = module.family_features(list(reversed(samples)), transformed, genetic_map, 100, model)
    actual = {(row['indID1'], row['indID2']): row for row in repeated}
    for row in original:
        pair = tuple(sorted([rename[row['indID1']], rename[row['indID2']]]))
        assert {key: value for key, value in actual[pair].items() if key not in ['indID1', 'indID2']} == pytest.approx(
            {key: value for key, value in row.items() if key not in ['indID1', 'indID2']})

def test_fast_components_match_original_with_duplicate_overlapping_and_conflicting_edges(monkeypatch):
    import itertools
    from pathlib import Path
    import numpy as np
    monkeypatch.syspath_prepend(str(Path(__file__).parents[1] / 'experiments/task10_development_2026_11_20'))
    from fast_multiway_features import ComponentCache
    original = _multiway_module()
    groups = [('a', 'b'), ('c',), ('d', 'e'), ('f', 'g')]
    samples = sorted(sample for group in groups for sample in group)
    genetic_map = {'1': (np.array([0., 50., 100.]), np.array([0., 35., 100.])),
        '2': (np.array([0., 100.]), np.array([0., 100.]))}
    for seed in range(20):
        rng = np.random.default_rng(seed)
        segments = {}
        for pair in itertools.combinations(samples, 2):
            chromosomes = {}
            for chromosome in genetic_map:
                rows = []
                for _ in range(int(rng.integers(0, 4))):
                    start, end = sorted(map(int, rng.choice(101, size=2, replace=False)))
                    row = (start, end, int(rng.integers(1, 3)), int(rng.integers(1, 3)))
                    rows.extend([row, row] if rng.random() < .3 else [row])
                if rows:
                    chromosomes[chromosome] = rows
            if chromosomes:
                segments[pair] = chromosomes
        cache = ComponentCache(groups, segments, genetic_map, 200.)
        for size in [2, 3]:
            for indices in itertools.combinations(range(len(groups)), size):
                for threshold in [5, 10]:
                    expected = original.component_integrals([groups[i] for i in indices], segments, genetic_map, 200., threshold)
                    assert cache.integral(indices, threshold) == pytest.approx(expected, rel=1e-12, abs=1e-12)

def test_fast_components_exhaustive_three_person_diploid_graphs(monkeypatch):
    import itertools
    from pathlib import Path
    import numpy as np
    monkeypatch.syspath_prepend(str(Path(__file__).parents[1] / 'experiments/task10_development_2026_11_20'))
    from fast_multiway_features import ComponentCache
    original = _multiway_module()
    groups = [('a',), ('b',), ('c',)]
    genetic_map = {'1': (np.array([0., 100.]), np.array([0., 100.]))}
    edges = [(pair, h1, h2) for pair in itertools.combinations(['a', 'b', 'c'], 2)
        for h1, h2 in itertools.product([1, 2], repeat=2)]
    for mask in range(1 << len(edges)):
        segments = {}
        for index, (pair, h1, h2) in enumerate(edges):
            if mask & (1 << index):
                segments.setdefault(pair, {}).setdefault('1', []).append((0, 100, h1, h2))
        expected = original.component_integrals(groups, segments, genetic_map, 100., 5)
        assert ComponentCache(groups, segments, genetic_map, 100.).integral([0, 1, 2], 5) == expected


def test_prospective_missingness_masks_are_nested_and_sample_order_invariant(tmp_path, monkeypatch):
    from pathlib import Path
    monkeypatch.syspath_prepend(str(Path(__file__).parents[1] / 'experiments/task10_development_2026_11_20'))
    from verify_missingness_contract import mask_genotypes
    sources = [tmp_path / 'forward.gz', tmp_path / 'reverse.gz']
    for source, names in zip(sources, [['a', 'b'], ['b', 'a']], strict=True):
        with gzip.open(source, 'wt') as handle:
            handle.write('#CHROM\tPOS\tID\tREF\tALT\tQUAL\tFILTER\tINFO\tFORMAT\t' + '\t'.join(names) + '\n')
            for position in range(1, 101):
                calls = ['0|1' if sample == 'a' else '1|1' for sample in names]
                handle.write(f'1\t{position}\t.\tA\tC\t.\tPASS\t.\tGT\t' + '\t'.join(calls) + '\n')
    outputs = [{rate: tmp_path / f'{side}_{rate}.gz' for rate in [.05, .20]} for side in ['f', 'r']]
    reports = [mask_genotypes(source, output, {'1': 76543}) for source, output in zip(sources, outputs, strict=True)]
    assert reports[0] == reports[1]
    masks = []
    for rate in [.05, .20]:
        calls = []
        for index, output in enumerate(outputs):
            with gzip.open(output[rate], 'rt') as handle:
                calls.append([line.rstrip().split('\t')[9:][::1 if index == 0 else -1]
                    for line in handle if not line.startswith('#')])
        assert calls[0] == calls[1]
        masks.append({(row, sample) for row, values in enumerate(calls[0]) for sample, call in enumerate(values) if call == '.|.'})
    assert masks[0] < masks[1]


def test_forest_mean_distance_encoding_and_validation(monkeypatch):
    from pathlib import Path
    import math
    monkeypatch.syspath_prepend(str(Path(__file__).parents[1] / 'experiments/task10_development_2026_11_20'))
    from forest_distance_decoder import model_distances
    from ghist_relatedness.labels import HUMAN_LABELS
    probabilities = dict.fromkeys(HUMAN_LABELS.values(), 0.)
    probabilities.update(full_1st_cousin=.5, full_2nd_cousin=.5)
    row = {'indID1': 'a', 'indID2': 'b', 'relation': 'full_1st_cousin', 'probabilities': probabilities}
    features, distances = model_distances([row])
    assert distances[0]['model_mean_meioses'] == 5.
    assert 1 - math.log2(features[0]['relatedness_ge_10']) == 5.
    with pytest.raises(ValueError, match='Duplicate'):
        model_distances([row, row])
    with pytest.raises(ValueError, match='sum to one'):
        model_distances([row | {'probabilities': dict.fromkeys(HUMAN_LABELS.values(), 0.)}])


def test_forest_distance_decoder_exact_tree_and_order_invariance(monkeypatch):
    from pathlib import Path
    import itertools
    monkeypatch.syspath_prepend(str(Path(__file__).parents[1] / 'experiments/task10_development_2026_11_20'))
    from forest_distance_decoder import decode
    from ghist_relatedness.labels import HUMAN_LABELS
    rows = []
    for a, b in itertools.combinations(['a', 'b', 'c', 'd'], 2):
        label = 'siblings' if (a, b) in [('a', 'b'), ('c', 'd')] else 'full_1st_cousin'
        probabilities = dict.fromkeys(HUMAN_LABELS.values(), 0.) | {label: 1.}
        rows.append({'indID1': a, 'indID2': b, 'relation': label, 'probabilities': probabilities})
    expected = [{key: row[key] for key in ['indID1', 'indID2', 'relation']} for row in rows]
    actual, report = decode(rows)
    flipped = [row | {'indID1': row['indID2'], 'indID2': row['indID1']} for row in reversed(rows)]
    reordered, _ = decode(flipped)
    assert actual == reordered == expected
    assert report['uncertainty_calibrated'] is False
    assert report['threshold_cm'] is None
